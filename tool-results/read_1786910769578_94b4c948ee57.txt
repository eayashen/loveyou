    14→function getSessionId(): string {
    15→  if (typeof window === "undefined") return "";
    16→  let id = sessionStorage.getItem("love_session_id");
    17→  if (!id) {
    18→    id = `sess_${Date.now()}_${Math.random().toString(36).slice(2, 10)}`;
    19→    sessionStorage.setItem("love_session_id", id);
    20→  }
    21→  return id;
    22→}

    24→async function trackVisit(payload: {
    25→  sessionId: string;
    26→  pageType: "login" | "chapter";
    27→  chapterId?: number | null;
    28→  chapterTitle?: string;
    29→  durationSeconds?: number;
    30→}) {
    31→  try {
    32→    await fetch("/api/track/visit", {
    33→      method: "POST",
    34→      headers: { "Content-Type": "application/json" },
    35→      body: JSON.stringify(payload),
    36→    });
    37→  } catch {
    38→    /* silent — tracking should never break the UI */
    39→  }
    40→}

    51→  }

   141→function LoginScreen({ onLogin }: { onLogin: () => void }) {
   142→  const [password, setPassword] = useState("");
   143→  const [showPassword, setShowPassword] = useState(false);
   144→  const [error, setError] = useState("");
   145→  const [isLoading, setIsLoading] = useState(false);
   146→  const [shake, setShake] = useState(false);
   147→  const inputRef = useRef<HTMLInputElement>(null);
   148→  const enterTimeRef = useRef<number>(Date.now());
   149→
   150→  useEffect(() => {
   151→    inputRef.current?.focus();
   152→    enterTimeRef.current = Date.now();
   153→    const sid = getSessionId();
   154→    trackVisit({ sessionId: sid, pageType: "login", durationSeconds: 0 });
   155→
   156→    return () => {
   157→      const duration = Math.round((Date.now() - enterTimeRef.current) / 1000);
   158→      trackVisit({ sessionId: sid, pageType: "login", durationSeconds: duration });
   159→    };
   160→  }, []);
   161→
   162→  const handleSubmit = useCallback(
   163→    async (e: React.FormEvent) => {
   164→      e.preventDefault();
   165→      if (!password.trim()) return;
   166→
   167→      setIsLoading(true);
   168→      setError("");
   169→
   170→      try {
   171→        const res = await fetch("/api/login", {
   172→          method: "POST",
   173→          headers: { "Content-Type": "application/json" },
   174→          body: JSON.stringify({ password }),
   175→        });
   176→        const data = await res.json();
   177→
   178→        if (data.success) {
   179→          const duration = Math.round((Date.now() - enterTimeRef.current) / 1000);
   180→          trackVisit({ sessionId: getSessionId(), pageType: "login", durationSeconds: duration });
   181→          onLogin();
   182→        } else {
   183→          setError(data.message);
   184→          setShake(true);
   185→          setTimeout(() => setShake(false), 600);
   186→        }
   187→      } catch {
   188→        setError("Network error, please try again");
   189→      } finally {
   190→        setIsLoading(false);
   191→      }
   192→    },
   193→    [password, onLogin]
   194→  );
   195→
   196→  return (
   197→    <div className="min-h-screen flex items-center justify-center relative overflow-hidden bg-gradient-to-br from-rose-50 via-pink-50 to-orange-50">
   198→      <FloatingPetals count={18} />
   199→
   200→      <motion.div
   201→        initial={{ opacity: 0, y: 30, scale: 0.95 }}
   202→        animate={{ opacity: 1, y: 0, scale: 1 }}
   203→        transition={{ duration: 0.8, ease: "easeOut" }}
   204→        className={`relative z-10 w-full max-w-sm mx-4 ${shake ? "animate-[shake_0.5s_ease-in-out]" : ""}`}
   205→      >
   206→        <div className="glass rounded-3xl p-8 shadow-xl shadow-rose-200/40 border border-rose-100/60">
   207→          {/* Hi text */}
   208→          <motion.div
   209→            className="flex justify-center mb-6"
   210→            animate={{ scale: [1, 1.08, 1] }}
   211→            transition={{ duration: 2, repeat: Infinity, ease: "easeInOut" }}
   212→          >
   213→            <div className="w-16 h-16 rounded-full bg-gradient-to-br from-rose-300 to-pink-400 flex items-center justify-center shadow-lg shadow-rose-300/50">
   214→              <span className="text-2xl text-white font-cursive font-bold">Hi</span>
   215→            </div>
   216→          </motion.div>
   217→
   218→          {/* Title */}
   219→          <div className="text-center mb-6">
   220→            <h1 className="font-cursive text-3xl text-rose-600 mb-2">
   221→              Just for You
   222→            </h1>
   223→            <p className="text-sm text-rose-400/80 font-serif">
   224→              Written with nothing but truth...
   225→            </p>
   226→          </div>
   227→
   228→          {/* Form */}
   229→          <form onSubmit={handleSubmit} className="space-y-4">
   230→            <div className="relative">
   231→              <Lock className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-rose-300" />
   232→              <Input
   233→                ref={inputRef}
   234→                type={showPassword ? "text" : "password"}
   235→                placeholder="Secret key.."
   236→                value={password}
   237→                onChange={(e) => {
   238→                  setPassword(e.target.value);
   239→                  setError("");
   240→                }}
   241→                className="pl-10 pr-10 h-12 rounded-2xl border-rose-200 bg-white/60 focus:bg-white/90 text-rose-800 placeholder:text-rose-300 font-serif text-sm"
   242→                disabled={isLoading}
   243→              />
   244→              <button
   245→                type="button"
   246→                onClick={() => setShowPassword(!showPassword)}
   247→                className="absolute right-3 top-1/2 -translate-y-1/2 text-rose-300 hover:text-rose-500 transition-colors"
   248→                tabIndex={-1}
   249→              >
   250→                {showPassword ? <EyeOff className="w-4 h-4" /> : <Eye className="w-4 h-4" />}
   251→              </button>
   252→            </div>
   253→
   254→            {error && (
   255→              <motion.p

   258→                className="text-sm text-rose-500 text-center font-serif"
   259→              >
   260→                {error}
   261→              </motion.p>
   262→            )}
   263→
   264→            <Button
   265→              type="submit"
   266→              disabled={isLoading || !password.trim()}
   267→              className="w-full h-12 rounded-2xl bg-gradient-to-r from-rose-400 to-pink-500 hover:from-rose-500 hover:to-pink-600 text-white font-cursive text-lg shadow-lg shadow-rose-300/40 transition-all duration-300 disabled:opacity-50"
   268→            >
   269→              {isLoading ? (
   270→                <motion.div
   271→                  animate={{ rotate: 360 }}
   272→                  transition={{ duration: 1, repeat: Infinity, ease: "linear" }}
   273→                  className="w-5 h-5 border-2 border-white/30 border-t-white rounded-full"
   274→                />
   275→              ) : (
   276→                "Decrypt >>>"
   277→              )}
   278→            </Button>
   279→          </form>
   280→        </div>
   281→
   282→
   283→      </motion.div>
   284→    </div>

   675→/* ------------------------------------------------------------------ */
   676→/*  CHAPTER PAGE                                                      */
   677→/* ------------------------------------------------------------------ */
   678→function StoryPage({ onSayYes }: { onSayYes: () => void }) {
   679→  const [currentChapter, setCurrentChapter] = useState(0);
   680→  const [direction, setDirection] = useState(1);
   681→  const sessionIdRef = useRef("");
   682→  const enterTimeRef = useRef<number>(Date.now());
   683→  const prevChapterRef = useRef(0);
   684→
   685→  // Init session
   686→  useEffect(() => {
   687→    sessionIdRef.current = getSessionId();
   688→    enterTimeRef.current = Date.now();
   689→  }, []);
   690→
   691→  const goToChapter = useCallback(
   692→    (index: number) => {
   693→      if (index < 0 || index >= chapters.length) return;
   694→
   695→      // Track the current chapter before leaving
   696→      const chapter = chapters[prevChapterRef.current];
   697→      const duration = Math.round((Date.now() - enterTimeRef.current) / 1000);
   698→      trackVisit({
   699→        sessionId: sessionIdRef.current,
   700→        pageType: "chapter",
   701→        chapterId: chapter.id,
   702→        chapterTitle: chapter.title,
   703→        durationSeconds: duration,
   704→      });
   705→
   706→      setDirection(index > currentChapter ? 1 : -1);
   707→      setCurrentChapter(index);
   708→      enterTimeRef.current = Date.now();
   709→      prevChapterRef.current = index;
   710→
   711→      // Scroll to top on chapter change
   712→      window.scrollTo({ top: 0, behavior: "instant" });
   713→    },
   714→    [currentChapter]
   715→  );
   716→
   717→  const goNext = useCallback(() => goToChapter(currentChapter + 1), [currentChapter, goToChapter]);
   718→  const goPrev = useCallback(() => goToChapter(currentChapter - 1), [currentChapter, goToChapter]);
   719→
   720→  // Track when leaving the page (chapter 1 initial visit)
   721→  useEffect(() => {
   722→    const sid = sessionIdRef.current;
   723→    return () => {
   724→      const chapter = chapters[prevChapterRef.current];
   725→      const duration = Math.round((Date.now() - enterTimeRef.current) / 1000);
   726→      trackVisit({
   727→        sessionId: sid,
   728→        pageType: "chapter",
   729→        chapterId: chapter.id,
   730→        chapterTitle: chapter.title,
   731→        durationSeconds: duration,
   732→      });
   733→    };
   734→  }, []);
   735→
   736→  const chapter = chapters[currentChapter];
   737→  const isProposal = currentChapter === chapters.length - 1;
   738→  const progress = ((currentChapter + 1) / chapters.length) * 100;
   739→
   740→  const slideVariants = {
   741→    enter: (dir: number) => ({ x: dir > 0 ? 300 : -300, opacity: 0 }),
   742→    center: { x: 0, opacity: 1 },
   743→    exit: (dir: number) => ({ x: dir > 0 ? -300 : 300, opacity: 0 }),
   744→  };
   745→
   746→  return (
   747→    <div
   748→      className={`min-h-screen bg-gradient-to-br ${chapter.bgGradient} relative overflow-hidden`}
   749→    >
   750→      <FloatingPetals count={10} />
   751→
   752→      {/* Progress bar */}
   753→      <div className="fixed top-0 left-0 right-0 z-50 h-1 bg-rose-100/50">
   754→        <motion.div
   755→          className="h-full bg-gradient-to-r from-rose-300 to-pink-400"
   756→          initial={false}
   757→          animate={{ width: `${progress}%` }}
   758→          transition={{ duration: 0.5, ease: "easeInOut" }}
   759→        />
   760→      </div>
   761→
   762→      {/* Chapter counter */}
   763→      <div className="fixed top-4 right-4 z-50 flex items-center gap-3">
   764→        <span className="text-xs text-rose-400/60 font-serif bg-white/50 backdrop-blur-sm px-3 py-1 rounded-full">
   765→          {currentChapter + 1} / {chapters.length}
   766→        </span>
   767→      </div>
   768→
   769→      {/* Main content area */}
   770→      <div className="min-h-screen flex flex-col items-center justify-center px-4 py-16">
   771→        <AnimatePresence mode="wait" custom={direction}>
   772→          <motion.div
   773→            key={currentChapter}
   774→            custom={direction}
   775→            variants={slideVariants}
   776→            initial="enter"
   777→            animate="center"
   778→            exit="exit"
   779→            transition={{ duration: 0.45, ease: "easeInOut" }}
   780→            className="w-full max-w-md mx-auto"
   781→          >
   782→            {/* Chapter header */}
   783→            <div className="text-center mb-8">
   784→              <motion.p
   785→                initial={{ opacity: 0, y: -10 }}
   786→                animate={{ opacity: 1, y: 0 }}
   787→                transition={{ delay: 0.2, duration: 0.5 }}
   788→                className="text-xs uppercase tracking-widest text-rose-400/60 font-serif mb-2"
   789→              >
   790→                Chapter {chapter.id}
   791→              </motion.p>
   792→              <motion.h2
   793→                initial={{ opacity: 0, y: -10 }}
   794→                animate={{ opacity: 1, y: 0 }}
   795→                transition={{ delay: 0.3, duration: 0.5 }}
   796→                className="font-cursive text-3xl md:text-4xl text-rose-600 mb-2"
   797→              >
   798→                {chapter.title}
   799→              </motion.h2>
   800→              <motion.p
   801→                initial={{ opacity: 0 }}
   802→                animate={{ opacity: 1 }}
   803→                transition={{ delay: 0.5, duration: 0.5 }}
   804→                className="text-sm text-rose-400/70 font-serif italic"
   805→              >
   806→                {chapter.subtitle}
   807→              </motion.p>
   808→            </div>
   809→
   810→            {/* Chapter image */}
   811→            <motion.div
   812→              initial={{ opacity: 0, scale: 0.9 }}
   813→              animate={{ opacity: 1, scale: 1 }}
   814→              transition={{ delay: 0.2, duration: 0.6 }}
   815→              className="relative w-full max-w-xs mx-auto mb-8 rounded-3xl overflow-hidden shadow-xl shadow-rose-200/30 aspect-[768/1344]"
   816→            >
   817→              <Image
   818→                src={chapter.image}
   819→                alt={chapter.title}
   820→                fill
   821→                className="object-cover"
   822→                priority={currentChapter < 2}
   823→              />
   824→              <div className="absolute inset-0 bg-gradient-to-t from-white/20 to-transparent" />
   825→            </motion.div>
   826→
   827→            {/* Chapter content */}
   828→            <motion.div
   829→              initial={{ opacity: 0, y: 20 }}
   830→              animate={{ opacity: 1, y: 0 }}
   831→              transition={{ delay: 0.4, duration: 0.6 }}
   832→            >
   833→              {isProposal ? (
   834→                <ProposalChapter sessionId={sessionIdRef.current} onSayYes={onSayYes} />
   835→              ) : (
   836→                <div className="glass rounded-3xl p-6 md:p-8 shadow-lg shadow-rose-100/20 border border-rose-100/30">
   837→                  {chapter.content}
   838→                </div>
   839→              )}
   840→            </motion.div>
   841→          </motion.div>
   842→        </AnimatePresence>
   843→
   844→        {/* Navigation */}
   845→        <div className="fixed bottom-0 left-0 right-0 z-50 pb-6 pt-8 bg-gradient-to-t from-white/60 via-white/30 to-transparent pointer-events-none">
   846→          <div className="flex items-center justify-center gap-6 max-w-md mx-auto px-4 pointer-events-auto">
   847→            <Button
   848→              onClick={goPrev}
   849→              disabled={currentChapter === 0}
   850→              variant="outline"
   851→              className="h-11 w-11 rounded-full border-rose-200 bg-white/70 backdrop-blur-sm text-rose-500 hover:bg-rose-50 hover:text-rose-600 hover:border-rose-300 disabled:opacity-30 disabled:cursor-not-allowed shadow-sm transition-all duration-300"
   852→              aria-label="Previous chapter"
   853→            >
   854→              <ChevronLeft className="w-5 h-5" />
   855→            </Button>
   856→
   857→            {/* Dots indicator */}
   858→            <div className="flex items-center gap-2">
   859→              {chapters.map((_, i) => (
   860→                <button
   861→                  key={i}
   862→                  onClick={() => goToChapter(i)}
   863→                  className={`rounded-full transition-all duration-300 ${
   864→                    i === currentChapter
   865→                      ? "w-6 h-2.5 bg-rose-400 shadow-sm shadow-rose-300/50"
   866→                      : "w-2.5 h-2.5 bg-rose-200/60 hover:bg-rose-300/80"
   867→                  }`}
   868→                  aria-label={`Go to chapter ${i + 1}`}
   869→                />
   870→              ))}
   871→            </div>
   872→
   873→            <Button
   874→              onClick={goNext}
   875→              disabled={currentChapter === chapters.length - 1}
   876→              variant="outline"
   877→              className="h-11 w-11 rounded-full border-rose-200 bg-white/70 backdrop-blur-sm text-rose-500 hover:bg-rose-50 hover:text-rose-600 hover:border-rose-300 disabled:opacity-30 disabled:cursor-not-allowed shadow-sm transition-all duration-300"
   878→              aria-label="Next chapter"
   879→            >
   880→              <ChevronRight className="w-5 h-5" />
   881→            </Button>
   882→          </div>
   883→        </div>
   884→      </div>
   885→    </div>
   886→  );
   887→}
