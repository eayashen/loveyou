    13→/* ------------------------------------------------------------------ */
    14→function getSessionId(): string {
    15→  if (typeof window === "undefined") return "";
    16→  let id = sessionStorage.getItem("love_session_id");
    17→  if (!id) {
    18→    id = `sess_${Date.now()}_${Math.random().toString(36).slice(2, 10)}`;
    19→    sessionStorage.setItem("love_session_id", id);
    20→  }
    21→  return id;
    22→}
    23→
    24→async function trackVisit(payload: {
    25→  sessionId: string;
    26→  pageType: "login" | "chapter";
    27→  chapterId?: number | null;
    28→  chapterTitle?: string;
    29→  durationSeconds?: number;
    30→}) {
    31→  try {
    32→    await fetch("/api/track/visit", {
    33→      method: "POST",
    34→      headers: { "Content-Type": "application/json" },
    35→      body: JSON.stringify(payload),
    36→    });
    37→  } catch {
    38→    /* silent — tracking should never break the UI */
    39→  }
    40→}
    41→
    42→async function trackMessage(sessionId: string, message: string) {
    43→  try {
    44→    await fetch("/api/track/message", {
    45→      method: "POST",
    46→      headers: { "Content-Type": "application/json" },
    47→      body: JSON.stringify({ sessionId, message }),
    48→    });
    49→  } catch {
    50→    /* silent */
    51→  }
    52→}

    65→function FloatingPetals({ count = 15 }: { count?: number }) {
    66→  const mounted = useHydrated();
    67→  const items = useMemo(
    68→    () =>
    69→      Array.from({ length: count }).map(() => ({
    70→        left: `${Math.random() * 100}%`,
    71→        duration: `${6 + Math.random() * 8}s`,
    72→        delay: `${Math.random() * 6}s`,
    73→        size: `${12 + Math.random() * 16}px`,
    74→      })),
    75→    [count]
    76→  );
    77→
    78→  if (!mounted) return <div className="fixed inset-0 pointer-events-none z-0" aria-hidden="true" />;
    79→
    80→  return (
    81→    <div className="fixed inset-0 pointer-events-none overflow-hidden z-0" aria-hidden="true">
    82→      {items.map((s, i) => (
    83→        <span
    84→          key={i}
    85→          className="absolute text-rose-200/40 animate-float-up"
    86→          style={{
    87→            left: s.left,
    88→            "--duration": s.duration,
    89→            "--delay": s.delay,
    90→            fontSize: s.size,
    91→          } as React.CSSProperties}
    92→        >
    93→          ✿
    94→        </span>
    95→      ))}
    96→    </div>
    97→  );
    98→}

   141→function LoginScreen({ onLogin }: { onLogin: () => void }) {
   142→  const [password, setPassword] = useState("");
   143→  const [showPassword, setShowPassword] = useState(false);
   144→  const [error, setError] = useState("");
   145→  const [isLoading, setIsLoading] = useState(false);
   146→  const [shake, setShake] = useState(false);
   147→  const inputRef = useRef<HTMLInputElement>(null);
   148→  const enterTimeRef = useRef<number>(Date.now());
   149→
   150→  useEffect(() => {
   151→    inputRef.current?.focus();
   152→    enterTimeRef.current = Date.now();
   153→    const sid = getSessionId();
   154→    trackVisit({ sessionId: sid, pageType: "login", durationSeconds: 0 });
   155→
   156→    return () => {
   157→      const duration = Math.round((Date.now() - enterTimeRef.current) / 1000);
   158→      trackVisit({ sessionId: sid, pageType: "login", durationSeconds: duration });
   159→    };
   160→  }, []);
   161→
   162→  const handleSubmit = useCallback(
   163→    async (e: React.FormEvent) => {
   164→      e.preventDefault();
   165→      if (!password.trim()) return;
   166→
   167→      setIsLoading(true);
   168→      setError("");
   169→
   170→      try {
   171→        const res = await fetch("/api/login", {
   172→          method: "POST",
   173→          headers: { "Content-Type": "application/json" },
   174→          body: JSON.stringify({ password }),
   175→        });
   176→        const data = await res.json();
   177→
   178→        if (data.success) {
   179→          const duration = Math.round((Date.now() - enterTimeRef.current) / 1000);
   180→          trackVisit({ sessionId: getSessionId(), pageType: "login", durationSeconds: duration });
   181→          onLogin();
   182→        } else {
   183→          setError(data.message);
   184→          setShake(true);
   185→          setTimeout(() => setShake(false), 600);
   186→        }
   187→      } catch {
   188→        setError("Network error, please try again");
   189→      } finally {
   190→        setIsLoading(false);
   191→      }
   192→    },
   193→    [password, onLogin]
   194→  );
   195→
   196→  return (
   197→    <div className="min-h-screen flex items-center justify-center relative overflow-hidden bg-gradient-to-br from-rose-50 via-pink-50 to-orange-50">
   198→      <FloatingPetals count={18} />
   199→
   200→      <motion.div
   201→        initial={{ opacity: 0, y: 30, scale: 0.95 }}
   202→        animate={{ opacity: 1, y: 0, scale: 1 }}
   203→        transition={{ duration: 0.8, ease: "easeOut" }}
   204→        className={`relative z-10 w-full max-w-sm mx-4 ${shake ? "animate-[shake_0.5s_ease-in-out]" : ""}`}
   205→      >
   206→        <div className="glass rounded-3xl p-8 shadow-xl shadow-rose-200/40 border border-rose-100/60">
   207→          {/* Hi text */}
   208→          <motion.div
   209→            className="flex justify-center mb-6"
   210→            animate={{ scale: [1, 1.08, 1] }}
   211→            transition={{ duration: 2, repeat: Infinity, ease: "easeInOut" }}
   212→          >
   213→            <div className="w-16 h-16 rounded-full bg-gradient-to-br from-rose-300 to-pink-400 flex items-center justify-center shadow-lg shadow-rose-300/50">
   214→              <span className="text-2xl text-white font-cursive font-bold">Hi</span>
   215→            </div>
   216→          </motion.div>
   217→
   218→          {/* Title */}
   219→          <div className="text-center mb-6">
   220→            <h1 className="font-cursive text-3xl text-rose-600 mb-2">
   221→              Just for You
   222→            </h1>
   223→            <p className="text-sm text-rose-400/80 font-serif">
   224→              Written with nothing but truth...
   225→            </p>
   226→          </div>
   227→
   228→          {/* Form */}
   229→          <form onSubmit={handleSubmit} className="space-y-4">
   230→            <div className="relative">
   231→              <Lock className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-rose-300" />
   232→              <Input
   233→                ref={inputRef}
   234→                type={showPassword ? "text" : "password"}
   235→                placeholder="Secret key.."
   236→                value={password}
   237→                onChange={(e) => {
   238→                  setPassword(e.target.value);
   239→                  setError("");
   240→                }}
   241→                className="pl-10 pr-10 h-12 rounded-2xl border-rose-200 bg-white/60 focus:bg-white/90 text-rose-800 placeholder:text-rose-300 font-serif text-sm"
   242→                disabled={isLoading}
   243→              />
   244→              <button
   245→                type="button"
   246→                onClick={() => setShowPassword(!showPassword)}
   247→                className="absolute right-3 top-1/2 -translate-y-1/2 text-rose-300 hover:text-rose-500 transition-colors"
   248→                tabIndex={-1}
   249→              >
   250→                {showPassword ? <EyeOff className="w-4 h-4" /> : <Eye className="w-4 h-4" />}
   251→              </button>
   252→            </div>
   253→
   254→            {error && (
   255→              <motion.p
   256→                initial={{ opacity: 0, y: -4 }}
   257→                animate={{ opacity: 1, y: 0 }}
   258→                className="text-sm text-rose-500 text-center font-serif"
   259→              >
   260→                {error}
   261→              </motion.p>
   262→            )}
   263→
   264→            <Button
   265→              type="submit"
   266→              disabled={isLoading || !password.trim()}
   267→              className="w-full h-12 rounded-2xl bg-gradient-to-r from-rose-400 to-pink-500 hover:from-rose-500 hover:to-pink-600 text-white font-cursive text-lg shadow-lg shadow-rose-300/40 transition-all duration-300 disabled:opacity-50"
   268→            >
   269→              {isLoading ? (
   270→                <motion.div
   271→                  animate={{ rotate: 360 }}
   272→                  transition={{ duration: 1, repeat: Infinity, ease: "linear" }}
   273→                  className="w-5 h-5 border-2 border-white/30 border-t-white rounded-full"
   274→                />
   275→              ) : (
   276→                "Decrypt >>>"
   277→              )}
   278→            </Button>
   279→          </form>
   280→        </div>
   281→
   282→
   283→      </motion.div>
   284→    </div>
   285→  );
   286→}

   689→function StoryPage({ onSayYes }: { onSayYes: () => void }) {
   690→  const [currentChapter, setCurrentChapter] = useState(0);
   691→  const [direction, setDirection] = useState(1);
   692→  const sessionIdRef = useRef("");
   693→  const enterTimeRef = useRef<number>(Date.now());
   694→  const prevChapterRef = useRef(0);
   695→
   696→  // Init session
   697→  useEffect(() => {
   698→    sessionIdRef.current = getSessionId();
   699→    enterTimeRef.current = Date.now();
   700→  }, []);
   701→
   702→  const goToChapter = useCallback(
   703→    (index: number) => {
   704→      if (index < 0 || index >= chapters.length) return;
   705→
   706→      // Track the current chapter before leaving
   707→      const chapter = chapters[prevChapterRef.current];
   708→      const duration = Math.round((Date.now() - enterTimeRef.current) / 1000);
   709→      trackVisit({
   710→        sessionId: sessionIdRef.current,
   711→        pageType: "chapter",
   712→        chapterId: chapter.id,
   713→        chapterTitle: chapter.title,
   714→        durationSeconds: duration,
   715→      });
   716→
   717→      setDirection(index > currentChapter ? 1 : -1);
   718→      setCurrentChapter(index);
   719→      enterTimeRef.current = Date.now();
   720→      prevChapterRef.current = index;
   721→
   722→      // Scroll to top on chapter change
   723→      window.scrollTo({ top: 0, behavior: "instant" });
   724→    },
   725→    [currentChapter]
   726→  );
   727→
   728→  const goNext = useCallback(() => goToChapter(currentChapter + 1), [currentChapter, goToChapter]);
   729→  const goPrev = useCallback(() => goToChapter(currentChapter - 1), [currentChapter, goToChapter]);
   730→
   731→  // Track when leaving the page (chapter 1 initial visit)
   732→  useEffect(() => {
   733→    const sid = sessionIdRef.current;
   734→    return () => {
   735→      const chapter = chapters[prevChapterRef.current];
   736→      const duration = Math.round((Date.now() - enterTimeRef.current) / 1000);
   737→      trackVisit({
   738→        sessionId: sid,
   739→        pageType: "chapter",
   740→        chapterId: chapter.id,
   741→        chapterTitle: chapter.title,
   742→        durationSeconds: duration,
   743→      });
   744→    };
   745→  }, []);
   746→
   747→  const chapter = chapters[currentChapter];
   748→  const isProposal = currentChapter === chapters.length - 1;
   749→  const progress = ((currentChapter + 1) / chapters.length) * 100;
   750→
   751→  const slideVariants = {
   752→    enter: (dir: number) => ({ x: dir > 0 ? 300 : -300, opacity: 0 }),
   753→    center: { x: 0, opacity: 1 },
   754→    exit: (dir: number) => ({ x: dir > 0 ? -300 : 300, opacity: 0 }),
   755→  };
   756→
   757→  return (
   758→    <div
   759→      className={`min-h-screen bg-gradient-to-br ${chapter.bgGradient} relative overflow-hidden`}

   761→      <FloatingPetals count={10} />

   763→      {/* Progress bar */}

   765→        <motion.div
   766→          className="h-full bg-gradient-to-r from-rose-300 to-pink-400"
   767→          initial={false}
   768→          animate={{ width: `${progress}%` }}
   769→          transition={{ duration: 0.5, ease: "easeInOut" }}
   770→        />
   771→      </div>
   772→
   773→      {/* Chapter counter */}
   774→      <div className="fixed top-4 right-4 z-50 flex items-center gap-3">
   775→        <span className="text-xs text-rose-400/60 font-serif bg-white/50 backdrop-blur-sm px-3 py-1 rounded-full">
   776→          {currentChapter + 1} / {chapters.length}
   777→        </span>
   778→      </div>
   779→
   780→      {/* Main content area */}
   781→      <div className="min-h-screen flex flex-col items-center justify-center px-4 py-16">
   782→        <AnimatePresence mode="wait" custom={direction}>

   788→            animate="center"
   789→            exit="exit"
   790→            transition={{ duration: 0.45, ease: "easeInOut" }}
   791→            className="w-full max-w-md mx-auto"
   792→          >
   793→            {/* Chapter header */}
   794→            <div className="text-center mb-8">
   795→              <motion.p
   796→                initial={{ opacity: 0, y: -10 }}
   797→                animate={{ opacity: 1, y: 0 }}
