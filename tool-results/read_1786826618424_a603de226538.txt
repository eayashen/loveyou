    14→    14→function getSessionId(): string {
    15→    15→  if (typeof window === "undefined") return "";
    16→    16→  let id = sessionStorage.getItem("love_session_id");
    17→    17→  if (!id) {
    18→    18→    id = `sess_${Date.now()}_${Math.random().toString(36).slice(2, 10)}`;
    19→    19→    sessionStorage.setItem("love_session_id", id);
    20→    20→  }
    21→    21→  return id;
    22→    22→}

    24→    24→async function trackVisit(payload: {
    25→    25→  sessionId: string;
    26→    26→  pageType: "login" | "chapter";
    27→    27→  chapterId?: number | null;
    28→    28→  chapterTitle?: string;

   141→   141→function LoginScreen({ onLogin }: { onLogin: () => void }) {
   142→   142→  const [password, setPassword] = useState("");
   143→   143→  const [showPassword, setShowPassword] = useState(false);
   144→   144→  const [error, setError] = useState("");
   145→   145→  const [isLoading, setIsLoading] = useState(false);
   146→   146→  const [shake, setShake] = useState(false);
   147→   147→  const inputRef = useRef<HTMLInputElement>(null);
   148→   148→  const enterTimeRef = useRef<number>(Date.now());
   149→   149→
   150→   150→  useEffect(() => {
   151→   151→    inputRef.current?.focus();
   152→   152→    enterTimeRef.current = Date.now();
   153→   153→    const sid = getSessionId();
   154→   154→    trackVisit({ sessionId: sid, pageType: "login", durationSeconds: 0 });
   155→   155→
   156→   156→    return () => {
   157→   157→      const duration = Math.round((Date.now() - enterTimeRef.current) / 1000);
   158→   158→      trackVisit({ sessionId: sid, pageType: "login", durationSeconds: duration });
   159→   159→    };
   160→   160→  }, []);
   161→   161→
   162→   162→  const handleSubmit = useCallback(
   163→   163→    async (e: React.FormEvent) => {
   164→   164→      e.preventDefault();
   165→   165→      if (!password.trim()) return;
   166→   166→
   167→   167→      setIsLoading(true);
   168→   168→      setError("");
   169→   169→
   170→   170→      try {
   171→   171→        const res = await fetch("/api/login", {
   172→   172→          method: "POST",
   173→   173→          headers: { "Content-Type": "application/json" },
   174→   174→          body: JSON.stringify({ password }),
   175→   175→        });
   176→   176→        const data = await res.json();
   177→   177→
   178→   178→        if (data.success) {
   179→   179→          const duration = Math.round((Date.now() - enterTimeRef.current) / 1000);
   180→   180→          trackVisit({ sessionId: getSessionId(), pageType: "login", durationSeconds: duration });
   181→   181→          onLogin();
   182→   182→        } else {
   183→   183→          setError(data.message);
   184→   184→          setShake(true);
   185→   185→          setTimeout(() => setShake(false), 600);
   186→   186→        }
   187→   187→      } catch {
   188→   188→        setError("Network error, please try again");
   189→   189→      } finally {
   190→   190→        setIsLoading(false);
   191→   191→      }
   192→   192→    },

   463→   463→function CelebrationPopup({ open, onClose }: { open: boolean; onClose: () => void }) {
   464→   464→  useEffect(() => {
   465→   465→    if (!open) return;
   466→   466→    const timer = setTimeout(() => {
   467→   467→      onClose();

   687→   687→  const enterTimeRef = useRef<number>(Date.now());
   688→   688→  const prevChapterRef = useRef(0);
   689→   689→
   690→   690→  // Init session
   691→   691→  useEffect(() => {
   692→   692→    sessionIdRef.current = getSessionId();
   693→   693→    enterTimeRef.current = Date.now();
   694→   694→  }, []);
   695→   695→
   696→   696→  const goToChapter = useCallback(
   697→   697→    (index: number) => {
   698→   698→      if (index < 0 || index >= chapters.length) return;
   699→   699→
   700→   700→      // Track the current chapter before leaving
   701→   701→      const chapter = chapters[prevChapterRef.current];
   702→   702→      const duration = Math.round((Date.now() - enterTimeRef.current) / 1000);
   703→   703→      trackVisit({
   704→   704→        sessionId: sessionIdRef.current,
   705→   705→        pageType: "chapter",
   706→   706→        chapterId: chapter.id,
   707→   707→        chapterTitle: chapter.title,
   708→   708→        durationSeconds: duration,
   709→   709→      });
   710→   710→
   711→   711→      setDirection(index > currentChapter ? 1 : -1);
   712→   712→      setCurrentChapter(index);
   713→   713→      enterTimeRef.current = Date.now();
