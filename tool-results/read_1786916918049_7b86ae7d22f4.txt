    13→    13→/* ------------------------------------------------------------------ */
    14→    14→function getSessionId(): string {
    15→    15→  if (typeof window === "undefined") return "";
    16→    16→  let id = sessionStorage.getItem("love_session_id");
    17→    17→  if (!id) {
    18→    18→    id = `sess_${Date.now()}_${Math.random().toString(36).slice(2, 10)}`;
    19→    19→    sessionStorage.setItem("love_session_id", id);
    20→    20→  }
    21→    21→  return id;
    22→    22→}

    24→    24→async function trackVisit(payload: {
    25→    25→  sessionId: string;
    26→    26→  pageType: "login" | "chapter";
    27→    27→  chapterId?: number | null;
    28→    28→  chapterTitle?: string;
    29→    29→  durationSeconds?: number;
    30→    30→}) {
    31→    31→  try {
    32→    32→    await fetch("/api/track/visit", {
    33→    33→      method: "POST",
    34→    34→      headers: { "Content-Type": "application/json" },
    35→    35→      body: JSON.stringify(payload),
    36→    36→    });
    37→    37→  } catch {
    38→    38→    /* silent — tracking should never break the UI */
    39→    39→  }
    40→    40→}

    49→    49→  } catch {
    50→    50→    /* silent */
    51→    51→  }
    52→    52→}

   141→   141→function LoginScreen({ onLogin }: { onLogin: () => void }) {
   142→   142→  const [password, setPassword] = useState("");
   143→   143→  const [showPassword, setShowPassword] = useState(false);
   144→   144→  const [error, setError] = useState("");
   145→   145→  const [isLoading, setIsLoading] = useState(false);
   146→   146→  const [shake, setShake] = useState(false);
   147→   147→  const inputRef = useRef<HTMLInputElement>(null);
   148→   148→  const enterTimeRef = useRef<number>(Date.now());
   149→   149→
   150→   150→  useEffect(() => {
   151→   151→    inputRef.current?.focus();
   152→   152→    enterTimeRef.current = Date.now();
   153→   153→    const sid = getSessionId();

   155→   155→
   156→   156→    return () => {
   157→   157→      const duration = Math.round((Date.now() - enterTimeRef.current) / 1000);
   158→   158→      trackVisit({ sessionId: sid, pageType: "login", durationSeconds: duration });
   159→   159→    };
   160→   160→  }, []);
   161→   161→
   162→   162→  const handleSubmit = useCallback(
   163→   163→    async (e: React.FormEvent) => {
   164→   164→      e.preventDefault();

   171→   171→        const res = await fetch("/api/login", {

   177→   177→
   178→   178→        if (data.success) {
   179→   179→          const duration = Math.round((Date.now() - enterTimeRef.current) / 1000);
   180→   180→          trackVisit({ sessionId: getSessionId(), pageType: "login", durationSeconds: duration });
   181→   181→          onLogin();
   182→   182→        } else {
   183→   183→          setError(data.message);
   184→   184→          setShake(true);
   185→   185→          setTimeout(() => setShake(false), 600);
   186→   186→        }
   187→   187→      } catch {
   188→   188→        setError("Network error, please try again");
   189→   189→      } finally {
   190→   190→        setIsLoading(false);
   191→   191→      }
   192→   192→    },
   193→   193→    [password, onLogin]
   194→   194→  );
   195→   195→

   204→   204→        className={`relative z-10 w-full max-w-sm mx-4 ${shake ? "animate-[shake_0.5s_ease-in-out]" : ""}`}
   205→   205→      >

   211→   211→            transition={{ duration: 2, repeat: Infinity, ease: "easeInOut" }}
   212→   212→          >

   214→   214→              <span className="text-2xl text-white font-cursive font-bold">Hi</span>
   215→   215→            </div>

   733→   733→function StoryPage() {

   735→   735→  const [direction, setDirection] = useState(1);
   736→   736→  const sessionIdRef = useRef("");
   737→   737→  const enterTimeRef = useRef<number>(Date.now());
   738→   738→  const prevChapterRef = useRef(0);
   739→   739→
   740→   740→  // Init session
   741→   741→  useEffect(() => {
   742→   742→    sessionIdRef.current = getSessionId();
   743→   743→    enterTimeRef.current = Date.now();
   744→   744→  }, []);
   745→   745→
   746→   746→  const goToChapter = useCallback(
   747→   747→    (index: number) => {
   748→   748→      if (index < 0 || index >= chapters.length) return;
   749→   749→
   750→   750→      // Track the current chapter before leaving
   751→   751→      const chapter = chapters[prevChapterRef.current];
   752→   752→      const duration = Math.round((Date.now() - enterTimeRef.current) / 1000);
   753→   753→      trackVisit({
   754→   754→        sessionId: sessionIdRef.current,
   755→   755→        pageType: "chapter",
   756→   756→        chapterId: chapter.id,
   757→   757→        chapterTitle: chapter.title,
   758→   758→        durationSeconds: duration,
   759→   759→      });
   760→   760→
   761→   761→      setDirection(index > currentChapter ? 1 : -1);
   762→   762→      setCurrentChapter(index);
   763→   763→      enterTimeRef.current = Date.now();
   764→   764→      prevChapterRef.current = index;
   765→   765→
   766→   766→      // Scroll to top on chapter change
   767→   767→      window.scrollTo({ top: 0, behavior: "instant" });

   769→   769→    [currentChapter]

   773→   773→  const goPrev = useCallback(() => goToChapter(currentChapter - 1), [currentChapter, goToChapter]);
   774→   774→
   775→   775→  // Track when leaving the page (chapter 1 initial visit)
