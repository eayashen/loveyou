    14→function getSessionId(): string {
    15→  if (typeof window === "undefined") return "";
    16→  let id = sessionStorage.getItem("love_session_id");
    17→  if (!id) {
    18→    id = `sess_${Date.now()}_${Math.random().toString(36).slice(2, 10)}`;
    19→    sessionStorage.setItem("love_session_id", id);
    20→  }
    21→  return id;
    22→}

    24→async function trackVisit(payload: {
    25→  sessionId: string;
    26→  pageType: "login" | "chapter";
    27→  chapterId?: number | null;
    28→  chapterTitle?: string;
    29→  durationSeconds?: number;
    30→}) {
    31→  try {
    32→    await fetch("/api/track/visit", {
    33→      method: "POST",
    34→      headers: { "Content-Type": "application/json" },
    35→      body: JSON.stringify(payload),
    36→    });
    37→  } catch {
    38→    /* silent — tracking should never break the UI */
    39→  }
    40→}

    49→  } catch {
    50→    /* silent */
    51→  }

   141→function LoginScreen({ onLogin }: { onLogin: () => void }) {
   142→  const [password, setPassword] = useState("");
   143→  const [showPassword, setShowPassword] = useState(false);
   144→  const [error, setError] = useState("");
   145→  const [isLoading, setIsLoading] = useState(false);
   146→  const [shake, setShake] = useState(false);
   147→  const inputRef = useRef<HTMLInputElement>(null);
   148→  const enterTimeRef = useRef<number>(Date.now());
   149→
   150→  useEffect(() => {
   151→    inputRef.current?.focus();
   152→    enterTimeRef.current = Date.now();
   153→    const sid = getSessionId();
   154→    trackVisit({ sessionId: sid, pageType: "login", durationSeconds: 0 });
   155→
   156→    return () => {
   157→      const duration = Math.round((Date.now() - enterTimeRef.current) / 1000);
   158→      trackVisit({ sessionId: sid, pageType: "login", durationSeconds: duration });
   159→    };
   160→  }, []);
   161→
   162→  const handleSubmit = useCallback(
   163→    async (e: React.FormEvent) => {
   164→      e.preventDefault();
   165→      if (!password.trim()) return;
   166→
   167→      setIsLoading(true);
   168→      setError("");
   169→
   170→      try {
   171→        const res = await fetch("/api/login", {
   172→          method: "POST",
   173→          headers: { "Content-Type": "application/json" },
   174→          body: JSON.stringify({ password }),
   175→        });
   176→        const data = await res.json();
   177→
   178→        if (data.success) {
   179→          const duration = Math.round((Date.now() - enterTimeRef.current) / 1000);
   180→          trackVisit({ sessionId: getSessionId(), pageType: "login", durationSeconds: duration });
   181→          onLogin();
   182→        } else {
   183→          setError(data.message);
   184→          setShake(true);
   185→          setTimeout(() => setShake(false), 600);
   186→        }
   187→      } catch {
   188→        setError("Network error, please try again");
   189→      } finally {
   190→        setIsLoading(false);
   191→      }
   192→    },
   193→    [password, onLogin]
   194→  );
   195→
   196→  return (
   197→    <div className="min-h-screen flex items-center justify-center relative overflow-hidden bg-gradient-to-br from-rose-50 via-pink-50 to-orange-50">
   198→      <FloatingPetals count={18} />
   199→
   200→      <motion.div
   201→        initial={{ opacity: 0, y: 30, scale: 0.95 }}
   202→        animate={{ opacity: 1, y: 0, scale: 1 }}
   203→        transition={{ duration: 0.8, ease: "easeOut" }}
   204→        className={`relative z-10 w-full max-w-sm mx-4 ${shake ? "animate-[shake_0.5s_ease-in-out]" : ""}`}
   205→      >
   206→        <div className="glass rounded-3xl p-8 shadow-xl shadow-rose-200/40 border border-rose-100/60">
   207→          {/* Hi text */}
   208→          <motion.div
   209→            className="flex justify-center mb-6"
   210→            animate={{ scale: [1, 1.08, 1] }}
   211→            transition={{ duration: 2, repeat: Infinity, ease: "easeInOut" }}
   212→          >
   213→            <div className="w-16 h-16 rounded-full bg-gradient-to-br from-rose-300 to-pink-400 flex items-center justify-center shadow-lg shadow-rose-300/50">
   214→              <span className="text-2xl text-white font-cursive font-bold">Hi</span>
   215→            </div>
   216→          </motion.div>
   217→
   218→          {/* Title */}
   219→          <div className="text-center mb-6">
   220→            <h1 className="font-cursive text-3xl text-rose-600 mb-2">
   221→              Just for You
   222→            </h1>
   223→            <p className="text-sm text-rose-400/80 font-serif">
   224→              This is something I made with all my heart.
   225→              <br />
   226→              Enter the secret word to continue...
   227→            </p>
   228→          </div>
   229→
   230→          {/* Form */}
   231→          <form onSubmit={handleSubmit} className="space-y-4">

   233→              <Lock className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-rose-300" />
   234→              <Input
   235→                ref={inputRef}
   236→                type={showPassword ? "text" : "password"}
   237→                placeholder="Your secret word..."
   238→                value={password}
   239→                onChange={(e) => {
   240→                  setPassword(e.target.value);
   241→                  setError("");
   242→                }}
   243→                className="pl-10 pr-10 h-12 rounded-2xl border-rose-200 bg-white/60 focus:bg-white/90 text-rose-800 placeholder:text-rose-300 font-serif text-sm"
   244→                disabled={isLoading}
   245→              />
   246→              <button
   247→                type="button"
   248→                onClick={() => setShowPassword(!showPassword)}
   249→                className="absolute right-3 top-1/2 -translate-y-1/2 text-rose-300 hover:text-rose-500 transition-colors"

   262→                {error}
   263→              </motion.p>
   264→            )}
   265→
   266→            <Button
   267→              type="submit"
   268→              disabled={isLoading || !password.trim()}
   269→              className="w-full h-12 rounded-2xl bg-gradient-to-r from-rose-400 to-pink-500 hover:from-rose-500 hover:to-pink-600 text-white font-cursive text-lg shadow-lg shadow-rose-300/40 transition-all duration-300 disabled:opacity-50"
   270→            >
   271→              {isLoading ? (
   272→                <motion.div
   273→                  animate={{ rotate: 360 }}
   274→                  transition={{ duration: 1, repeat: Infinity, ease: "linear" }}
   275→                  className="w-5 h-5 border-2 border-white/30 border-t-white rounded-full"
   276→                />
   277→              ) : (
   278→                "Open My Heart"
   279→              )}
   280→            </Button>
   281→          </form>
   282→        </div>
   283→
   284→        {/* Footer text */}

   286→          Made with love, just for you

   528→function ProposalChapter({ sessionId, onSayYes }: { sessionId: string; onSayYes: () => void }) {
   529→  const [showPopup, setShowPopup] = useState(false);
   530→  const [message, setMessage] = useState("");
   531→  const messageDebounceRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   532→
   533→  const handleYes = () => {
   534→    setShowPopup(true);
   535→  };
   536→
   537→  const handlePopupClose = useCallback(() => {
   538→    setShowPopup(false);
   539→  }, []);
   540→
   541→  // Debounced message save
   542→  const handleMessageChange = useCallback(
   543→    (value: string) => {
   544→      setMessage(value);
   545→      if (messageDebounceRef.current) clearTimeout(messageDebounceRef.current);
   546→      if (value.trim()) {
   547→        messageDebounceRef.current = setTimeout(() => {
   548→          trackMessage(sessionId, value);
   549→        }, 1500);
   550→      }
   551→    },
   552→    [sessionId]
   553→  );
   554→
   555→  // Save message on unmount if there's content
   556→  useEffect(() => {
   557→    return () => {
   558→      if (messageDebounceRef.current) clearTimeout(messageDebounceRef.current);
   559→    };
   560→  }, []);
   561→
   562→  return (
   563→    <div className="relative">
   564→      <SparklesOverlay />
   565→
   566→      <div className="space-y-8 text-center">
   567→        <motion.div
   568→          initial={{ opacity: 0, scale: 0.8 }}
   569→          animate={{ opacity: 1, scale: 1 }}
   570→          transition={{ duration: 0.8, delay: 0.2 }}

   680→/* ------------------------------------------------------------------ */
   681→/*  CHAPTER PAGE                                                      */
   682→/* ------------------------------------------------------------------ */
   683→function StoryPage({ onSayYes }: { onSayYes: () => void }) {
   684→  const [currentChapter, setCurrentChapter] = useState(0);
   685→  const [direction, setDirection] = useState(1);
   686→  const sessionIdRef = useRef("");
   687→  const enterTimeRef = useRef<number>(Date.now());
   688→  const prevChapterRef = useRef(0);
   689→
   690→  // Init session
   691→  useEffect(() => {
   692→    sessionIdRef.current = getSessionId();
   693→    enterTimeRef.current = Date.now();
   694→  }, []);
   695→
   696→  const goToChapter = useCallback(
   697→    (index: number) => {
   698→      if (index < 0 || index >= chapters.length) return;
   699→
   700→      // Track the current chapter before leaving
   701→      const chapter = chapters[prevChapterRef.current];
   702→      const duration = Math.round((Date.now() - enterTimeRef.current) / 1000);
   703→      trackVisit({
   704→        sessionId: sessionIdRef.current,
   705→        pageType: "chapter",
   706→        chapterId: chapter.id,
   707→        chapterTitle: chapter.title,
   708→        durationSeconds: duration,
   709→      });
   710→
   711→      setDirection(index > currentChapter ? 1 : -1);
   712→      setCurrentChapter(index);
   713→      enterTimeRef.current = Date.now();
   714→      prevChapterRef.current = index;
   715→
   716→      // Scroll to top on chapter change
   717→      window.scrollTo({ top: 0, behavior: "instant" });
   718→    },
   719→    [currentChapter]
   720→  );
   721→
   722→  const goNext = useCallback(() => goToChapter(currentChapter + 1), [currentChapter, goToChapter]);
   723→  const goPrev = useCallback(() => goToChapter(currentChapter - 1), [currentChapter, goToChapter]);
   724→
   725→  // Track when leaving the page (chapter 1 initial visit)
   726→  useEffect(() => {
   727→    const sid = sessionIdRef.current;
   728→    return () => {
   729→      const chapter = chapters[prevChapterRef.current];
   730→      const duration = Math.round((Date.now() - enterTimeRef.current) / 1000);
   731→      trackVisit({
   732→        sessionId: sid,
   733→        pageType: "chapter",
   734→        chapterId: chapter.id,
   735→        chapterTitle: chapter.title,
   736→        durationSeconds: duration,
   737→      });
   738→    };
   739→  }, []);
   740→
   741→  const chapter = chapters[currentChapter];
   742→  const isProposal = currentChapter === chapters.length - 1;
   743→  const progress = ((currentChapter + 1) / chapters.length) * 100;
   744→
   745→  const slideVariants = {
   746→    enter: (dir: number) => ({ x: dir > 0 ? 300 : -300, opacity: 0 }),
   747→    center: { x: 0, opacity: 1 },
   748→    exit: (dir: number) => ({ x: dir > 0 ? -300 : 300, opacity: 0 }),
   749→  };
   750→
   751→  return (
   752→    <div
   753→      className={`min-h-screen bg-gradient-to-br ${chapter.bgGradient} relative overflow-hidden`}
   754→    >
   755→      <FloatingPetals count={10} />
   756→
   757→      {/* Progress bar */}
   758→      <div className="fixed top-0 left-0 right-0 z-50 h-1 bg-rose-100/50">
   759→        <motion.div
   760→          className="h-full bg-gradient-to-r from-rose-300 to-pink-400"
   761→          initial={false}
   762→          animate={{ width: `${progress}%` }}
   763→          transition={{ duration: 0.5, ease: "easeInOut" }}
   764→        />
   765→      </div>
   766→
   767→      {/* Chapter counter */}
   768→      <div className="fixed top-4 right-4 z-50 flex items-center gap-3">
   769→        <span className="text-xs text-rose-400/60 font-serif bg-white/50 backdrop-blur-sm px-3 py-1 rounded-full">
   770→          {currentChapter + 1} / {chapters.length}
   771→        </span>
   772→      </div>
   773→
   774→      {/* Main content area */}
   775→      <div className="min-h-screen flex flex-col items-center justify-center px-4 py-16">
   776→        <AnimatePresence mode="wait" custom={direction}>
   777→          <motion.div
   778→            key={currentChapter}
   779→            custom={direction}
   780→            variants={slideVariants}
   781→            initial="enter"
   782→            animate="center"
   783→            exit="exit"
   784→            transition={{ duration: 0.45, ease: "easeInOut" }}
   785→            className="w-full max-w-md mx-auto"
   786→          >
   787→            {/* Chapter header */}
   788→            <div className="text-center mb-8">
   789→              <motion.p
   790→                initial={{ opacity: 0, y: -10 }}
   791→                animate={{ opacity: 1, y: 0 }}
   792→                transition={{ delay: 0.2, duration: 0.5 }}
   793→                className="text-xs uppercase tracking-widest text-rose-400/60 font-serif mb-2"
   794→              >
   795→                Chapter {chapter.id}
   796→              </motion.p>
   797→              <motion.h2
   798→                initial={{ opacity: 0, y: -10 }}
   799→                animate={{ opacity: 1, y: 0 }}
   800→                transition={{ delay: 0.3, duration: 0.5 }}
   801→                className="font-cursive text-3xl md:text-4xl text-rose-600 mb-2"
   802→              >
   803→                {chapter.title}
   804→              </motion.h2>
   805→              <motion.p
   806→                initial={{ opacity: 0 }}
   807→                animate={{ opacity: 1 }}
   808→                transition={{ delay: 0.5, duration: 0.5 }}
   809→                className="text-sm text-rose-400/70 font-serif italic"
   810→              >
   811→                {chapter.subtitle}
   812→              </motion.p>
   813→            </div>
   814→
   815→            {/* Chapter image */}
   816→            <motion.div
   817→              initial={{ opacity: 0, scale: 0.9 }}
   818→              animate={{ opacity: 1, scale: 1 }}
   819→              transition={{ delay: 0.2, duration: 0.6 }}
   820→              className="relative w-full max-w-xs mx-auto mb-8 rounded-3xl overflow-hidden shadow-xl shadow-rose-200/30 aspect-[768/1344]"
   821→            >
   822→              <Image
   823→                src={chapter.image}
   824→                alt={chapter.title}
   825→                fill
   826→                className="object-cover"
   827→                priority={currentChapter < 2}
   828→              />
   829→              <div className="absolute inset-0 bg-gradient-to-t from-white/20 to-transparent" />
   830→            </motion.div>
   831→
   832→            {/* Chapter content */}
   833→            <motion.div
   834→              initial={{ opacity: 0, y: 20 }}
   835→              animate={{ opacity: 1, y: 0 }}
   836→              transition={{ delay: 0.4, duration: 0.6 }}
   837→            >
   838→              {isProposal ? (
   839→                <ProposalChapter sessionId={sessionIdRef.current} onSayYes={onSayYes} />
   840→              ) : (
   841→                <div className="glass rounded-3xl p-6 md:p-8 shadow-lg shadow-rose-100/20 border border-rose-100/30">
   842→                  {chapter.content}
   843→                </div>
   844→              )}
   845→            </motion.div>
   846→          </motion.div>
   847→        </AnimatePresence>
   848→
   849→        {/* Navigation */}
   850→        <div className="fixed bottom-0 left-0 right-0 z-50 pb-6 pt-8 bg-gradient-to-t from-white/60 via-white/30 to-transparent pointer-events-none">
   851→          <div className="flex items-center justify-center gap-6 max-w-md mx-auto px-4 pointer-events-auto">
   852→            <Button
   853→              onClick={goPrev}
   854→              disabled={currentChapter === 0}
   855→              variant="outline"
   856→              className="h-11 w-11 rounded-full border-rose-200 bg-white/70 backdrop-blur-sm text-rose-500 hover:bg-rose-50 hover:text-rose-600 hover:border-rose-300 disabled:opacity-30 disabled:cursor-not-allowed shadow-sm transition-all duration-300"
   857→              aria-label="Previous chapter"
   858→            >
   859→              <ChevronLeft className="w-5 h-5" />
   860→            </Button>
   861→
   862→            {/* Dots indicator */}
   863→            <div className="flex items-center gap-2">
   864→              {chapters.map((_, i) => (
   865→                <button
   866→                  key={i}
   867→                  onClick={() => goToChapter(i)}
   868→                  className={`rounded-full transition-all duration-300 ${
   869→                    i === currentChapter
   870→                      ? "w-6 h-2.5 bg-rose-400 shadow-sm shadow-rose-300/50"
   871→                      : "w-2.5 h-2.5 bg-rose-200/60 hover:bg-rose-300/80"
   872→                  }`}
   873→                  aria-label={`Go to chapter ${i + 1}`}
   874→                />
   875→              ))}
   876→            </div>
   877→
   878→            <Button
   879→              onClick={goNext}
   880→              disabled={currentChapter === chapters.length - 1}
   881→              variant="outline"
   882→              className="h-11 w-11 rounded-full border-rose-200 bg-white/70 backdrop-blur-sm text-rose-500 hover:bg-rose-50 hover:text-rose-600 hover:border-rose-300 disabled:opacity-30 disabled:cursor-not-allowed shadow-sm transition-all duration-300"
   883→              aria-label="Next chapter"
   884→            >
   885→              <ChevronRight className="w-5 h-5" />
   886→            </Button>
   887→          </div>
   888→        </div>
   889→      </div>
   890→    </div>
   891→  );
   892→}
